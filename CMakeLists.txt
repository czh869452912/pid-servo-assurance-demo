cmake_minimum_required(VERSION 3.20)
project(pid_servo C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(pid OBJECT src/pid.c)
target_include_directories(pid PUBLIC include)
if(MSVC)
  target_compile_options(pid PRIVATE /W4 /WX)
else()
  target_compile_options(pid PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_executable(test_pid tests/test_pid.c)
target_link_libraries(test_pid PRIVATE pid)
foreach(case IN ITEMS RANGE P I D SAT AW RESET INVALID)
  add_test(NAME REQ-PID-${case} COMMAND test_pid ${case})
endforeach()
